const since = state.get("since");
const mail = tool("list_emails", { account: "work", since });
const real = mail.messages.filter(m => !m.unsubscribe);
if (real.length === 0) return "nothing new";

const todo = llm({
  tier: "light",
  prompt: "Which of these need a reply from me? One line each.",
  input: real,
});

output.inbox({ title: `Mail: ${todo.count} need a reply`, body: todo.text });
state.set("since", mail.messages[0]?.date ?? since);
